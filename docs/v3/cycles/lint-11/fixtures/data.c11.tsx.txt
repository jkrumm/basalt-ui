// seed data-primitive/chart-data-fallback 23: the fallback hidden in a helper the prop calls

// seed data-primitive/timer-refetch 21: an async interval that refetches, then bumps a counter
export function useAsyncTick(refetch: () => Promise<unknown>) {
  const [, setTicks] = useState(0)
  useEffect(() => {
    const id = setInterval(async () => {
      await refetch()
      setTicks((t) => t + 1)
    }, 5_000)
    return () => clearInterval(id)
  }, [refetch])
}

// seed data-primitive/timer-refetch 22: a hook handed the refetch itself
export function useByHook(refetch: () => unknown) {
  useIntervalEffect(refetch, 30_000)
}

// seed data-primitive/timer-refetch 23: an interval that refetches only while the tab is visible
export function useVisibleTick(refetch: () => unknown) {
  useEffect(() => {
    const id = setInterval(() => {
      if (!document.hidden) refetch()
    }, 15_000)
    return () => clearInterval(id)
  }, [refetch])
}

// pass data-primitive/timer-refetch 4: a debounce timer that only moves state
export function useDebounced(q: string) {
  const [debounced, setDebounced] = useState(q)
  useEffect(() => {
    const id = setTimeout(() => setDebounced(q), 300)
    return () => clearTimeout(id)
  }, [q])
  return debounced
}

// seed data-primitive/unseamed-transport 21: a WebSocket opened inside a useSyncExternalStore subscribe
export function useSocketStatus(url: string) {
  const subscribe = (notify: () => void) => {
    const ws = new WebSocket(url)
    ws.onopen = notify
    ws.onclose = notify
    return () => ws.close()
  }
  return useSyncExternalStore(subscribe, () => true, () => false)
}
