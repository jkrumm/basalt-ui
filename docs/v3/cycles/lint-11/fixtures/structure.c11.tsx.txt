// seed primitive-bypass/svg-data-marks 22: a radar polygon whose points are joined from the data
export function Radar({ rows }: { rows: Row[] }) {
  const points = rows.map((r, i) => `${50 + r.value * Math.cos(i)},${50 + r.value * Math.sin(i)}`)
  return (
    <svg viewBox="0 0 100 100">
      <polygon points={points.join(' ')} />
    </svg>
  )
}
