// seed data-primitive/reader-loop 11: a reader taken through optional chaining and read in a loop
export async function tail(url: string, on: (b: Uint8Array) => void) {
  const res = await fetch(url)
  const reader = res.body?.getReader()
  while (reader) {
    const r = await reader.read()
    if (r.done) break
    on(r.value)
  }
}

// seed data-primitive/reader-loop 12: a decoded stream piped to a writable sink
export async function sink(url: string, on: (s: string) => void) {
  const res = await fetch(url)
  await res.body!.pipeThrough(new TextDecoderStream()).pipeTo(new WritableStream({ write: on }))
}
