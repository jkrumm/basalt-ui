// seed data-primitive/reader-loop 17: SSE events through eventsource-parser's stream after the decoder
import { EventSourceParserStream } from 'eventsource-parser/stream'
export async function events(res: Response, on: (data: string) => void) {
  const stream = res.body!.pipeThrough(new TextDecoderStream()).pipeThrough(new EventSourceParserStream())
  for await (const event of stream) on(event.data)
}

// seed data-primitive/reader-loop 18: a generator over a body handed in as a parameter
export async function* lines(body: ReadableStream<Uint8Array>) {
  for await (const chunk of body.pipeThrough(new TextDecoderStream())) yield chunk
}

// seed data-primitive/reader-loop 19: the decoded body bound with let
export async function readLet(res: Response, on: (s: string) => void) {
  let text = res.body!.pipeThrough(new TextDecoderStream())
  for await (const chunk of text) on(chunk)
  text = new ReadableStream()
}

// seed data-primitive/reader-loop 20: the decoded body through optional chaining, guarded
export async function readGuarded(res: Response, on: (s: string) => void) {
  const text = res.body?.pipeThrough(new TextDecoderStream())
  if (!text) return
  for await (const chunk of text) on(chunk)
}

// pass data-primitive/reader-loop 1: a dropped CSV file read as text, no network
export async function readCsv(file: File, on: (s: string) => void) {
  for await (const chunk of file.stream().pipeThrough(new TextDecoderStream())) on(chunk)
}

// pass data-primitive/reader-loop 2: a download's bytes counted for a progress bar
export async function download(url: string, onProgress: (n: number) => void) {
  const res = await fetch(url)
  let received = 0
  for await (const chunk of res.body!) {
    received += chunk.byteLength
    onProgress(received)
  }
}
