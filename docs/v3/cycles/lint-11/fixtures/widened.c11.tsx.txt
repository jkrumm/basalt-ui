// seed data-primitive/reader-loop 21: a helper destructuring the body out of the response it is handed
export const readEvents = async ({ body }: Response, onLine: (s: string) => void) => {
  for await (const chunk of body!.pipeThrough(new TextDecoderStream())) onLine(chunk)
}

// seed data-primitive/reader-loop 22: a parameter typed as the response's body, read by a reader loop
export async function pump(body: Response['body'], onChunk: (c: Uint8Array) => void) {
  const reader = body!.getReader()
  for (;;) {
    const { done, value } = await reader.read()
    if (done) return
    onChunk(value)
  }
}

// seed data-primitive/reader-loop 23: a client class whose method loops over a stream it is handed

  }
}

// pass data-primitive/reader-loop 4: a parameter named stream that is an async iterable of rows, no byte stream
export async function drain(stream: AsyncIterable<Run>) {

  return out
}
