// Seeded offenders for basalt/viewport-keyed. Each `// seed` line opens one phrasing; it ends at the next.
import { useEffect, useState } from 'react'
import { useMediaQuery } from '@base-ui/react/unstable-use-media-query'

// seed viewport-keyed/breakpoint-prefix 1: a two-column grid from md up, as shadcn blocks write it
export function StatsGrid() {
  return <div className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-4">stats</div>
}

// seed viewport-keyed/breakpoint-prefix 2: hide the sidebar below lg with a max- variant
export function Sidebar() {
  return <aside className="flex flex-col max-lg:hidden">nav</aside>
}

// seed viewport-keyed/arbitrary-query 1: an exact pixel breakpoint
export function Toolbar() {
  return <div className="flex flex-col min-[900px]:flex-row">tools</div>
}

// seed viewport-keyed/media-query 1: a media-query hook picks the layout
export function Layout() {
  const wide = useMediaQuery('(min-width: 1024px)')
  return <div className={wide ? 'flex' : 'grid'}>body</div>
}

// seed viewport-keyed/media-query 2: window.matchMedia in an effect, the hand-rolled useIsMobile
export function useIsMobile() {
  const [mobile, setMobile] = useState(false)
  useEffect(() => {
    const mq = window.matchMedia('(max-width: 767px)')
    setMobile(mq.matches)
  }, [])
  return mobile
}

// seed viewport-keyed/media-query 3: a window.innerWidth read decides the column count
export function Columns() {
  const cols = window.innerWidth < 768 ? 1 : 3
  return <div style={{ gridTemplateColumns: `repeat(${cols}, 1fr)` }}>cols</div>
}
