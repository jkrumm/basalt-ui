// Lint cycle 4 (cycle 3's scorer). Run from the scratch app as `bun score.ts <lint-output-file>`.
// A mark spans from its comment to the next mark. A `seed` mark is an offender: caught when its own
// rule reports a line inside the span. A `pass` mark is valid code: silent when its own rule reports
// nothing there, a false alarm when it does. Other ids that fired in the span are listed beside it.
import { readdirSync, readFileSync } from 'node:fs'
import { join } from 'node:path'

const SEEDS = 'src/seeds'
const findings = readFileSync(process.argv[2] ?? 'lint.txt', 'utf8')
  .split('\n')
  .flatMap((line) => {
    const m = /^(\S+?):(\d+) (\S+?): (.*)$/.exec(line)
    return m ? [{ file: m[1]!, line: Number(m[2]), id: m[3]!, text: m[4]! }] : []
  })

const ruleOf = (key: string): string => {
  const [head] = key.split('/')
  if (key.startsWith('stylelint/')) return key.split(' ')[0]!
  return `basalt/${head}`
}

const rows: string[] = []
const passRows: string[] = []
let caught = 0
let total = 0
let silent = 0
let passes = 0
for (const name of readdirSync(SEEDS).toSorted()) {
  const file = join(SEEDS, name)
  const lines = readFileSync(file, 'utf8').split('\n')
  const marks = lines.flatMap((text, i) => {
    const m = /(?:\/\/|\/\*) (seed|pass) (\S+) (\d+): (.*?)(?: \*\/)?$/.exec(text)
    return m ? [{ at: i + 1, kind: m[1]!, key: m[2]!, n: m[3]!, what: m[4]! }] : []
  })
  marks.forEach((mark, k) => {
    const end = marks[k + 1]?.at ?? lines.length + 1
    const inSpan = findings.filter((f) => f.file === file && f.line > mark.at && f.line < end)
    const want = ruleOf(mark.key)
    // In a stylesheet, raw-scroll is reported by the stylelint value list with its scroll message.
    const cssScroll = (f: (typeof inSpan)[number]): boolean =>
      want === 'basalt/raw-scroll' &&
      name.endsWith('.css') &&
      f.id === 'stylelint/declaration-property-value-disallowed-list' &&
      /scroll/.test(f.text)
    const hit = inSpan.find((f) => f.id === want || cssScroll(f))
    const others = [...new Set(inSpan.filter((f) => f.id !== want).map((f) => f.id))]
    if (mark.kind === 'pass') {
      passes++
      if (!hit) silent++
      passRows.push(
        `| ${mark.key} ${mark.n} | ${mark.what} | ${hit ? 'false alarm' : 'silent'} | ${hit ? `${name}:${hit.line}` : name} | ${others.join(', ') || '—'} |`,
      )
      return
    }
    total++
    if (hit) caught++
    rows.push(
      `| ${mark.key} ${mark.n} | ${mark.what} | ${hit ? 'caught' : 'missed'} | ${hit ? `${name}:${hit.line}` : `${name}`} | ${others.join(', ') || '—'} |`,
    )
  })
}
console.log('| Seed | Phrasing | Result | Line | Other ids in the span |')
console.log('|-|-|-|-|-|')
console.log(rows.join('\n'))
console.log(`\n${caught} of ${total} caught\n`)
console.log('| Valid code | Shape | Result | Line | Other ids in the span |')
console.log('|-|-|-|-|-|')
console.log(passRows.join('\n'))
console.log(`\n${silent} of ${passes} silent`)
