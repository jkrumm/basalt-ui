import { useEffect, useState } from 'react'
import { cn } from 'basalt-ui'
import { useMediaQuery } from '@base-ui/react/unstable-use-media-query'

const MOBILE_QUERY = '(max-width: 767px)'

// seed viewport-keyed/media-query 4: the query hoisted into a named constant, as a hooks file writes it
export function useCompact() {
  return useMediaQuery(MOBILE_QUERY)
}

// seed viewport-keyed/media-query 5: a resize listener comparing the width to a named breakpoint
const BREAKPOINT = 768
export function useNarrow() {
  const [narrow, setNarrow] = useState(false)
  useEffect(() => {
    const on = () => setNarrow(window.innerWidth < BREAKPOINT)
    window.addEventListener('resize', on)
    return () => window.removeEventListener('resize', on)
  }, [])
  return narrow
}
