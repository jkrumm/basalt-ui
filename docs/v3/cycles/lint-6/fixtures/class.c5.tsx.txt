// seed viewport-keyed/media-query 12: matchMedia behind useSyncExternalStore, the React 18 idiom
const subscribe = (cb: () => void) => {
  const mq = window.matchMedia('(max-width: 767px)')
  mq.addEventListener('change', cb)
  return () => mq.removeEventListener('change', cb)
}
export function useIsMobile() {
  return useSyncExternalStore(subscribe, () => window.matchMedia('(max-width: 767px)').matches)
}

// seed viewport-keyed/media-query 13: the document's client width compared to a breakpoint
export function useNarrowDoc() {
  return document.documentElement.clientWidth < 768
}

// seed viewport-keyed/media-query 14: the visual viewport's width read in an effect
export function useTight() {
  const [tight, setTight] = useState(false)
  useEffect(() => setTight((window.visualViewport?.width ?? 0) < 640), [])
  return tight
}
