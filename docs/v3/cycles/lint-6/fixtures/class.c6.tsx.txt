import { useEffect, useState } from 'react'
import { cn } from 'basalt-ui'
import { useWindowSize } from 'usehooks-ts'

// seed viewport-keyed/media-query 15: the outer window width against a breakpoint
export function useNarrowWindow() {
  const [narrow, setNarrow] = useState(false)
  useEffect(() => setNarrow(window.outerWidth < 768), [])
  return narrow
}

// seed viewport-keyed/media-query 16: a window-size hook's width compared to a breakpoint
export function Toolbar() {
  const { width = 0 } = useWindowSize()
  return width < 768 ? <span>menu</span> : <span>bar</span>
}

// pass viewport-keyed/media-query 1: the visual viewport's height for the phone keyboard inset
export function useKeyboardOpen() {
  const [open, setOpen] = useState(false)
  useEffect(() => {
    setOpen((window.visualViewport?.height ?? window.innerHeight) < window.innerHeight)
  }, [])
  return open
}
