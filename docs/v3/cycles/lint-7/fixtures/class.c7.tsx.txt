// seed viewport-keyed/media-query 17: the screen's width against a breakpoint
export function useCompact() {
  const [compact, setCompact] = useState(false)
  useEffect(() => setCompact(window.screen.width < 768), [])
  return compact
}

// seed viewport-keyed/media-query 18: an orientation query in JS, the form cycle 6 widened in classes
export function useStacked() {
  const [stacked, setStacked] = useState(false)
  useEffect(() => setStacked(window.matchMedia('(orientation: portrait)').matches), [])
  return stacked
}

// seed viewport-keyed/media-query 19: a user-agent sniff choosing the phone layout
export function Layout() {

  return phone ? <span>tabs</span> : <span>rail</span>
}
