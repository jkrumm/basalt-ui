import { useEffect, useState } from 'react'
import { useMediaQuery } from 'react-responsive'
import { cva } from 'class-variance-authority'
import { cn } from 'basalt-ui'

// seed viewport-keyed/media-query 20: react-responsive's object query, the hook an agent installs
export function useNarrow() {
  return useMediaQuery({ maxWidth: 767 })
}

// seed viewport-keyed/media-query 21: the visual viewport's width against a breakpoint
export function useSheetMode() {
  const [sheet, setSheet] = useState(false)
  useEffect(() => setSheet((window.visualViewport?.width ?? 0) < 640), [])
  return sheet
}

// seed viewport-keyed/media-query 22: the screen orientation API choosing a layout
export function useLandscape() {
  return window.screen.orientation.type.startsWith('landscape')
}

// seed viewport-keyed/media-query 23: a user-agent read that picks a shortcut glyph, which Kbd's Mod owns
export function ShortcutHint() {
  const mac = /Mac|iPhone|iPad/.test(navigator.userAgent)
  return <kbd>{mac ? '⌘K' : 'Ctrl K'}</kbd>
}

// pass viewport-keyed/media-query 4: a user-preference query in JS
export function useCalm() {
  return window.matchMedia('(prefers-reduced-motion: reduce)').matches
}

// seed viewport-keyed/breakpoint-prefix 15: a viewport class handed down through an item-class prop
