type Message = { id: string; text: string; html: string }

// seed control-home/outside-home 6: a Slider threshold in a card body on local state
export function Threshold() {
  const [v, setV] = useState(50)
  return (
    <Card>
      <Slider value={v} onValueChange={(n) => setV(Number(n))} />
    </Card>
  )
}

// seed control-home/outside-home 7: a RadioGroup in the toolbar primitive
export function Density() {
  const [d, setD] = useState('compact')
  return (
    <Toolbar>
      <RadioGroup value={d} onValueChange={(x) => setD(String(x))}>
        <Radio value="compact" />
      </RadioGroup>
    </Toolbar>
  )
}

// seed control-home/outside-home 8: a Combobox owner filter inside a local wrapper component

// seed untrusted-markdown/dangerous-html 3: innerHTML assigned from model text in an effect
export function useInject(el: HTMLElement | null, m: Message) {
  if (el) el.innerHTML = m.html
}

// seed untrusted-markdown/dangerous-html 4: insertAdjacentHTML with model html
export function append(el: HTMLElement, m: Message) {
  el.insertAdjacentHTML('beforeend', m.html)
}

// seed untrusted-markdown/rehype-raw 2: rehype-raw passed inline to the trusted Markdown
import rehypeRaw from 'rehype-raw'
export function Doc({ m }: { m: Message }) {
  return (
    <Markdown contentTrust="trusted" rehypePlugins={[rehypeRaw]}>
      {m.text}
    </Markdown>
  )
}

// seed untrusted-markdown/rehype-raw 3: rehype-raw loaded by a dynamic import
export const loadRaw = () => import('rehype-raw')

// seed untrusted-markdown/mdx-evaluate 3: MDX compiled and run with new Function
import { compile } from '@mdx-js/mdx'
export async function compileRun(source: string) {
  const code = String(await compile(source, { outputFormat: 'function-body' }))
  return new Function(code)
}

// seed untrusted-markdown/mdx-evaluate 4: next-mdx-remote for model MDX
import { MDXRemote } from 'next-mdx-remote/rsc'
export const R = MDXRemote

// seed untrusted-markdown/trusted-content 4: trusted chosen by a ternary on the role
export function Bubble({ m, role }: { m: Message; role: 'user' | 'assistant' }) {
  return <Markdown contentTrust={role === 'assistant' ? 'trusted' : 'untrusted'}>{m.text}</Markdown>
}

// seed untrusted-markdown/trusted-content 5: trusted spread from a props object
const trusted = { contentTrust: 'trusted' as const }
export function Answer({ m }: { m: Message }) {
  return <Markdown {...trusted}>{m.text}</Markdown>
}
