// seed data-primitive/timer-refetch 5: usehooks-ts useInterval that invalidates

export function useAutoRefresh() {

}

// seed data-primitive/timer-refetch 6: an interval calling a named refresh function
export function useTicker() {
  const qc = useQueryClient()
  useEffect(() => {
    const refresh = () => qc.invalidateQueries({ queryKey: ['runs'] })
    const id = setInterval(refresh, 5000)
    return () => clearInterval(id)
  }, [qc])
}

// seed data-primitive/reader-loop 4: a decoded stream read in a while loop
export async function readAll(res: Response) {
  const reader = res.body!.pipeThrough(new TextDecoderStream()).getReader()
  while (true) {
    const { done } = await reader.read()
    if (done) break
  }
}
