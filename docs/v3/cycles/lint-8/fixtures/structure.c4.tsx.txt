// seed data-primitive/reader-loop 5: a stream piped into a WritableStream sink
export async function pipe(res: Response, onChunk: (s: string) => void) {
  await res.body!.pipeThrough(new TextDecoderStream()).pipeTo(new WritableStream({ write: onChunk }))
}

// seed data-primitive/reader-loop 6: a reader drained with a while loop over a decoded stream
export async function drain(res: Response) {
  const reader = res.body!.pipeThrough(new TextDecoderStream()).getReader()
  let out = ''
  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    out += value
  }
  return out
}

// seed data-primitive/timer-refetch 7: a recursive timeout that invalidates
export function usePoll() {
  const qc = useQueryClient()
  useEffect(() => {
    let t: ReturnType<typeof setTimeout>
    const tick = () => {
      void qc.invalidateQueries({ queryKey: ['runs'] })
      t = setTimeout(tick, 5000)
    }
    tick()
    return () => clearTimeout(t)
  }, [qc])
}

// seed data-primitive/timer-refetch 8: window.setInterval calling refetch
export function useRefetch() {
  const q = useQuery(runsQuery)
  useEffect(() => {
    const id = window.setInterval(() => void q.refetch(), 5000)
    return () => window.clearInterval(id)
  }, [q])
}
