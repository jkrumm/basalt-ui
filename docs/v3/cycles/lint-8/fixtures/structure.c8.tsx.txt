// seed data-primitive/reader-loop 13: a response body iterated with for await
export async function readAll(res: Response) {
  let text = ''
  if (!res.body) return text
  for await (const chunk of res.body) text += String(chunk)
  return text
}

// seed data-primitive/reader-loop 14: a decoded stream's reader taken from pipeThrough
export async function readDecoded(res: Response) {
  const reader = res.body!.pipeThrough(new TextDecoderStream()).getReader()
  while (true) {
    const { done } = await reader.read()
    if (done) break
  }
}
