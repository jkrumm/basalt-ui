// seed viewport-keyed/media-query 24: usehooks-ts' window size against a breakpoint
export function useCompact() {
  const { width } = useWindowSize()
  return width < 768
}

// seed viewport-keyed/media-query 25: the client-hints mobile flag choosing a layout
export function useHandheld() {
  return navigator.userAgentData?.mobile === true
}

// seed viewport-keyed/media-query 26: a resize listener storing innerWidth for a layout switch
export function useColumns() {
  const [cols, setCols] = useState(3)
  useEffect(() => {
    const on = () => setCols(window.innerWidth > 1100 ? 3 : 1)
    window.addEventListener('resize', on)
    return () => window.removeEventListener('resize', on)
  }, [])
  return cols
}
