// seed data-primitive/reader-loop 9: a response body read as an async iterable
export async function stream(url: string, on: (s: string) => void) {
  const res = await fetch(url)
  const dec = new TextDecoder()
  for await (const chunk of res.body as unknown as AsyncIterable<Uint8Array>) on(dec.decode(chunk))
}

// seed data-primitive/reader-loop 10: a decoded stream read through pipeThrough and a reader
export async function stream2(url: string, on: (s: string) => void) {
  const res = await fetch(url)
  const reader = res.body!.pipeThrough(new TextDecoderStream()).getReader()
  for (;;) {
    const { done, value } = await reader.read()
    if (done) break
    on(value)
  }
}

// seed untrusted-markdown/dangerous-html 13: model html assigned to outerHTML
export function Outer({ r }: { r: Row }) {
  const ref = useRef<HTMLDivElement>(null)
  useEffect(() => {
    if (ref.current) ref.current.outerHTML = r.html
  }, [r.html])
  return <div ref={ref} />
}

// seed untrusted-markdown/dangerous-html 14: a contextual fragment from model html
export function Frag({ r }: { r: Row }) {
  const ref = useRef<HTMLDivElement>(null)
  useEffect(() => {
    ref.current?.append(document.createRange().createContextualFragment(r.html))
  }, [r.html])
  return <div ref={ref} />
}

// seed untrusted-markdown/dangerous-html 15: DOMParser output adopted into the page
export function Parsed({ r }: { r: Row }) {
  const ref = useRef<HTMLDivElement>(null)
  useEffect(() => {
    const doc = new DOMParser().parseFromString(r.html, 'text/html')
    ref.current?.replaceChildren(...Array.from(doc.body.childNodes))
  }, [r.html])
  return <div ref={ref} />
}
