### 2.3 The configs and the tag step that were proven

```jsonc
// master, pre-GA (.releaserc.json; plugins unchanged)
```

```bash
# publish.yml replacement for the last step (version from package.json, never from the ref: workflow_dispatch runs on a branch)
V=$(jq -r .version package.json); NAME=$(jq -r .name package.json)
if [[ "$V" == *-* ]]; then pre="${V#*-}"; DIST_TAG="${pre%%.*}"            # 3.0.0-next.4 -> next
else
  L=$(npm view --prefer-online "$NAME" dist-tags.latest 2>/dev/null || true)
  if [ -z "$L" ] || [ "$(printf '%s\n%s\n' "$L" "$V" | sort -V | tail -1)" = "$V" ]; then DIST_TAG=latest
  else DIST_TAG="latest-${V%%.*}"; fi                                          # a lower stable never retakes latest
fi
npm publish --access public --tag "$DIST_TAG"
```
